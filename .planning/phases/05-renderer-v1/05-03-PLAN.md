```python
from aerocloud.models.geometry import PlacedWord, DroppedWord, PlacementResult, PlacementStats, AABB, GlyphBBox, DropReason
import numpy as np
import torch

# 3 placed words with known positions
placed = [
    PlacedWord(word="hello", y=16, x=16, bbox=AABB(y_min=12, x_min=10, y_max=20, x_max=22), size_pt=24),
    PlacedWord(word="world", y=32, x=32, bbox=AABB(y_min=28, x_min=26, y_max=36, x_max=38), size_pt=24),
    PlacedWord(word="test", y=48, x=48, bbox=AABB(y_min=44, x_min=42, y_max=52, x_max=54), size_pt=24),
]
dropped = [DroppedWord(word="dropped", reason=DropReason.OUTSIDE_MASK)]
stats = PlacementStats(total_words=4, placed=3, dropped=1, total_iterations=100, wall_clock_ms=50.0)
result = PlacementResult(placements=placed, dropped_words=dropped, stats=stats)

# Create matching GlyphBBox for each placed word (D-06: only placed words enter renderer)
glyphs = []
for pw in placed:
    buf = np.random.default_rng(42).integers(0, 256, (8, 12), dtype=np.uint8)
    glyph = GlyphBBox(
        bbox=pw.bbox, advance_width=12, pixel_buffer=buf,
        font_family="Inter", size_pt=pw.size_pt, codepoint=ord(pw.word[0]),
    )
    glyphs.append(glyph)

# Build params from PlacementResult (D-06)
params = torch.tensor([[pw.y, pw.x, 1.0, 0.0] for pw in result.placements], dtype=torch.float32)
sprites = [register_glyph(g.font_family, g.size_pt, g.codepoint, g.pixel_buffer, torch.device("cpu")) for g in glyphs]
renderer = DifferentiableRenderer(params, sprites, torch.device("cpu"))
out = renderer(64, 64)
```
