```python
import numpy as np
import structlog
from sklearn.metrics.pairwise import cosine_similarity

from aerocloud.semantic.errors import SemanticError

logger = structlog.get_logger(__name__)

def cosine_similarity_matrix(embeddings: np.ndarray) -> np.ndarray:
    """Compute pairwise cosine similarity for (N, 384) embeddings.

    Args:
        embeddings: (N, D) float32 array where D is embedding dimension.

    Returns:
        (N, N) float32 symmetric matrix with 1.0 on diagonal.

    Raises:
        SemanticError: If embeddings is empty (N=0).
    """
    if embeddings.ndim != 2 or embeddings.shape[0] == 0:
        raise SemanticError(f"embeddings must be (N, D) with N>0, got {embeddings.shape}")
    sim = cosine_similarity(embeddings).astype(np.float32)
    logger.debug("semantic.cosine.computed", n=embeddings.shape[0], shape=sim.shape)
    return sim
```

```python
from __future__ import annotations

from typing import Literal

import numpy as np
import structlog

from aerocloud.config import settings
from aerocloud.semantic.errors import ProjectionError

logger = structlog.get_logger(__name__)

def project_to_2d(
    embeddings: np.ndarray,
    method: Literal["umap", "tsne"] | None = None,
    seed: int | None = None,
) -> np.ndarray:
    """Project (N, D) embeddings to (N, 2) using UMAP or t-SNE.

    Args:
        embeddings: (N, D) float32 array.
        method: Projection method. Defaults to settings.projection_method.
        seed: Random seed for reproducibility. Defaults to settings.seed.

    Returns:
        (N, 2) float32 array of 2D positions.

    Raises:
        ProjectionError: If method is invalid or projection fails.
    """
    method = method or settings.projection_method
    seed = seed if seed is not None else settings.seed
    n = embeddings.shape[0]

    if method == "umap":
        import umap
        # Clamp n_neighbors to N-1 for small inputs (Claude's discretion)
        n_neighbors = min(15, max(2, n - 1))
        reducer = umap.UMAP(
            n_components=2,
            n_neighbors=n_neighbors,
            min_dist=0.1,
            metric="cosine",
            random_state=seed,
        )
        coords = reducer.fit_transform(embeddings.astype(np.float32))
    elif method == "tsne":
        from sklearn.manifold import TSNE
        perplexity = min(30.0, max(1.0, float(n - 1)))
        reducer = TSNE(
            n_components=2,
            perplexity=perplexity,
            random_state=seed,
            init="pca",
        )
        coords = reducer.fit_transform(embeddings.astype(np.float32))
    else:
        raise ProjectionError(f"Unknown projection method: {method!r}. Use 'umap' or 'tsne'.")

    result = np.asarray(coords, dtype=np.float32)
    logger.debug("semantic.projection.complete", method=method, n=n, output_shape=result.shape)
    return result
```
