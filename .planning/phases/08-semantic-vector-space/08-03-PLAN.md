```python
from __future__ import annotations

import numpy as np
import ot
import structlog

from aerocloud.config import settings
from aerocloud.models.semantic import TransportPlan
from aerocloud.semantic.errors import SinkhornNonConvergenceError

logger = structlog.get_logger(__name__)

def compute_transport(
    cost_matrix: np.ndarray,
    eps_init: float | None = None,
    max_iter: int | None = None,
) -> TransportPlan:
    """Compute optimal transport plan via Sinkhorn-Knopp in log-space (D-07).

    Uses uniform marginals: a = b = ones(N)/N (D-08 word-to-position).
    Adaptive epsilon per D-09: halve if niter < 10, double if niter > 500.
    Epsilon clamped to [1e-4, 1.0].

    Convergence: O(log(1/eps)) independent of dimension (arXiv:2604.03787, D-10).

    Args:
        cost_matrix: (N, N) float array of pairwise costs (e.g., cosine distances).
        eps_init: Initial regularization. Defaults to settings.sinkhorn_eps_init.
        max_iter: Max Sinkhorn iterations. Defaults to settings.sinkhorn_max_iter.

    Returns:
        TransportPlan with (N, N) doubly stochastic transport_matrix, eps_used, iterations.

    Raises:
        SinkhornNonConvergenceError: If Sinkhorn did not converge and output contains NaN.
        SemanticError: If cost_matrix is empty or non-square.
    """
    eps = eps_init if eps_init is not None else settings.sinkhorn_eps_init
    max_iter = max_iter if max_iter is not None else settings.sinkhorn_max_iter

    # Clamp initial eps to valid range
    eps = float(np.clip(eps, 1e-4, 1.0))

    N = cost_matrix.shape[0]
    if N == 0 or cost_matrix.ndim != 2 or cost_matrix.shape[0] != cost_matrix.shape[1]:
        from aerocloud.semantic.errors import SemanticError
        raise SemanticError(f"cost_matrix must be square (N, N) with N>0, got {cost_matrix.shape}")

    a = np.ones(N, dtype=np.float64) / N
    b = np.ones(N, dtype=np.float64) / N
    M = cost_matrix.astype(np.float64)

    T, log_dict = ot.sinkhorn(
        a, b, M,
        reg=eps,
        method="sinkhorn_log",
        numItermax=max_iter,
        log=True,
    )

    niter = log_dict.get("niter", max_iter)

    if np.any(np.isnan(T)):
        raise SinkhornNonConvergenceError(
            f"Sinkhorn produced NaN after {niter} iterations with eps={eps}"
        )

    # D-09 adaptive adjustment for NEXT call (returned for caller to reuse)
    eps_adapted = eps
    if niter < 10:
        eps_adapted = max(1e-4, eps / 2.0)
    elif niter > 500:
        eps_adapted = min(1.0, eps * 2.0)

    logger.info(
        "semantic.transport.complete",
        n=N, eps_init=eps, eps_adapted=eps_adapted, niter=niter,
    )

    return TransportPlan(
        transport_matrix=T,
        eps_used=eps_adapted,
        iterations=niter,
    )
```

```python
import numpy as np
from hypothesis import given, settings as h_settings
from hypothesis import strategies as st

from aerocloud.semantic.transport import compute_transport

@given(
    n=st.integers(min_value=2, max_value=20),
    seed=st.integers(min_value=0, max_value=999),
)
@h_settings(max_examples=50, deadline=30_000)
def test_transport_is_doubly_stochastic(n: int, seed: int) -> None:
    rng = np.random.default_rng(seed)
    cost = rng.uniform(0.0, 1.0, size=(n, n)).astype(np.float64)
    plan = compute_transport(cost, eps_init=0.1, max_iter=1000)
    T = plan.transport_matrix
    row_sums = T.sum(axis=1)
    col_sums = T.sum(axis=0)
    expected = np.ones(n) / n
    np.testing.assert_allclose(row_sums, expected, atol=1e-3)
    np.testing.assert_allclose(col_sums, expected, atol=1e-3)
    assert np.all(T >= -1e-10), "Transport matrix has negative values"
```
