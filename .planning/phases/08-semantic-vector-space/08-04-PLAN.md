```python
from __future__ import annotations

import numpy as np
import structlog
import torch

from aerocloud.config import settings
from aerocloud.geometry.mat import MATResult
from aerocloud.models.tokens import WordCandidate
from aerocloud.semantic.cosine import cosine_similarity_matrix
from aerocloud.semantic.embeddings import encode_surfaces
from aerocloud.semantic.projection import project_to_2d
from aerocloud.semantic.transport import compute_transport

logger = structlog.get_logger(__name__)

def semantic_warm_start(
    candidates: list[WordCandidate],
    sdf: np.ndarray,
    mat_result: MATResult,
    config: dict[str, object] | None = None,
) -> torch.Tensor:
    """Compute semantically-informed initial positions for DifferentiableRenderer (D-11).

    Pipeline:
    1. Encode candidate surfaces → (N, 384) embeddings
    2. Cosine distance matrix → cost matrix for transport
    3. UMAP project embeddings → (N, 2) semantic positions
    4. Build target positions from MAT branch origins (+ UMAP-scaled fill)
    5. Sinkhorn transport: assign words to positions
    6. Map assigned positions to SDF canvas coordinates (D-12)
    7. Return (N, 4) tensor [y, x, 1.0, 0.0]

    Args:
        candidates: List of N WordCandidates from NLP pipeline.
        sdf: (H, W) float32 SDF array (positive = inside, ADR-0004).
        mat_result: MATResult with branch origins for target positions.
        config: Optional overrides (eps_init, max_iter, projection_method, seed).

    Returns:
        (N, 4) float32 tensor [y, x, scale, theta] with requires_grad=False.
        Scale=1.0, theta=0.0 per D-11.
    """
    N = len(candidates)
    seed = config.get("seed", settings.seed) if config else settings.seed
    method = config.get("projection_method", settings.projection_method) if config else settings.projection_method

    surfaces = [c.surface for c in candidates]

    # Step 1: BERT encode
    embeddings = encode_surfaces(surfaces)  # (N, 384)

    # Step 2: Cosine similarity → distance cost matrix
    sim_matrix = cosine_similarity_matrix(embeddings)  # (N, N)
    cost_matrix = 1.0 - sim_matrix  # cosine distance

    # Step 3: UMAP/t-SNE project embeddings to 2D
    coords_2d = project_to_2d(embeddings, method=method, seed=seed)  # (N, 2)

    # Step 4: Build target positions from MAT branch origins
    # Use branch origins as anchor points, fill remaining with
    # UMAP-scaled positions within SDF positive region
    target_positions = _build_target_positions(coords_2d, sdf, mat_result, N)

    # Step 5: Sinkhorn transport
    # Cost between UMAP word positions and target canvas positions
    from scipy.spatial.distance import cdist
    transport_cost = cdist(coords_2d, target_positions, metric="euclidean").astype(np.float64)

    eps_init = config.get("eps_init", settings.sinkhorn_eps_init) if config else settings.sinkhorn_eps_init
    plan = compute_transport(transport_cost, eps_init=eps_init)

    # Step 6: Assign positions via transport matrix argmax per word
    T = plan.transport_matrix  # (N, N)
    assignments = T.argmax(axis=1)  # each word → best position
    assigned_positions = target_positions[assignments]  # (N, 2)

    # Step 7: Build (N, 4) params tensor [y, x, scale, theta]
    params = np.zeros((N, 4), dtype=np.float32)
    params[:, 0] = assigned_positions[:, 0]  # y
    params[:, 1] = assigned_positions[:, 1]  # x
    params[:, 2] = 1.0  # scale (D-11)
    params[:, 3] = 0.0  # theta (D-11)

    result = torch.from_numpy(params)  # requires_grad=False by default
    logger.info("semantic.warm_start.complete", n=N, eps=plan.eps_used, iterations=plan.iterations)
    return result


def _build_target_positions(
    coords_2d: np.ndarray,
    sdf: np.ndarray,
    mat_result: MATResult,
    n_positions: int,
) -> np.ndarray:
    """Build N target positions on the SDF canvas from MAT branches + UMAP fill (D-12).

    Strategy:
    1. Start with MAT branch origin_yx points (deepest interior points)
    2. If N > num_branches, add UMAP-projected positions scaled to SDF interior
    3. All positions clamped to SDF > 0 region

    Returns:
        (N, 2) float32 array of (y, x) canvas positions.
    """
    H, W = sdf.shape
    positions = []

    # MAT branch origins as primary anchor points
    for branch in mat_result.branches:
        positions.append(np.array(branch.origin_yx, dtype=np.float32))

    # Scale UMAP 2D coords to fit within SDF interior
    umap_min = coords_2d.min(axis=0)
    umap_max = coords_2d.max(axis=0)
    umap_range = umap_max - umap_min
    umap_range = np.where(umap_range < 1e-8, 1.0, umap_range)  # avoid div by zero

    # Find SDF interior bounding box
    inside_yx = np.argwhere(sdf > 0)
    if len(inside_yx) == 0:
        # Fallback: center of canvas
        inside_min = np.array([H / 4, W / 4], dtype=np.float32)
        inside_max = np.array([3 * H / 4, 3 * W / 4], dtype=np.float32)
    else:
        inside_min = inside_yx.min(axis=0).astype(np.float32)
        inside_max = inside_yx.max(axis=0).astype(np.float32)

    # Scale UMAP coords to SDF interior
    scaled = (coords_2d - umap_min) / umap_range  # [0, 1]
    interior_range = inside_max - inside_min
    canvas_coords = scaled * interior_range + inside_min  # map to SDF interior

    # Fill remaining positions from scaled UMAP coords
    for i in range(len(positions), n_positions):
        idx = i % len(canvas_coords)
        positions.append(canvas_coords[idx])

    # Truncate if too many MAT branches
    positions = positions[:n_positions]

    result = np.array(positions, dtype=np.float32)

    # Clamp all positions to valid SDF interior
    result[:, 0] = np.clip(result[:, 0], 0, H - 1)
    result[:, 1] = np.clip(result[:, 1], 0, W - 1)

    # Snap to nearest SDF > 0 pixel if position is outside
    for i in range(len(result)):
        y, x = int(result[i, 0]), int(result[i, 1])
        if y < H and x < W and sdf[y, x] <= 0:
            # Find nearest interior pixel
            dist_to_inside = np.where(sdf > 0, 0.0, np.inf)
            if len(inside_yx) > 0:
                dists = np.sqrt(((inside_yx - np.array([y, x])) ** 2).sum(axis=1))
                nearest_idx = dists.argmin()
                result[i] = inside_yx[nearest_idx].astype(np.float32)

    return result
```
