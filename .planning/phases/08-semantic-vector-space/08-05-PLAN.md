**Step 1 — Alembic migration (`0002_word_embeddings.py`):**

Per D-13 (table schema) and D-14 (HNSW index):

```python
"""Add word_embeddings table for BERT embedding cache (Phase 8 SEM-07).

Revision ID: 0002
Revises: 0001
"""
from alembic import op
import sqlalchemy as sa

revision = "0002"
down_revision = "0001"

def upgrade() -> None:
    # Ensure pgvector extension (idempotent, created in 0001)
    op.execute("CREATE EXTENSION IF NOT EXISTS vector")

    op.create_table(
        "word_embeddings",
        sa.Column("surface", sa.Text(), primary_key=True),
        sa.Column("embedding", sa.Column("vector(384)")),  # pgvector type
        sa.Column("model_version", sa.Text(), nullable=False, server_default="all-MiniLM-L6-v2"),
        sa.Column("created_at", sa.DateTime(timezone=True), server_default=sa.func.now()),
    )

    # HNSW index for cosine similarity queries (D-14)
    op.execute(
        "CREATE INDEX ix_word_embeddings_embedding_hnsw "
        "ON word_embeddings USING hnsw (embedding vector_cosine_ops)"
    )

def downgrade() -> None:
    op.execute("DROP INDEX IF EXISTS ix_word_embeddings_embedding_hnsw")
    op.drop_table("word_embeddings")
```

**Step 2 — Persistence module (`semantic/persistence.py`):**

```python
from __future__ import annotations

import numpy as np
import structlog
import asyncpg

from aerocloud.config import settings
from aerocloud.semantic.errors import SemanticError

logger = structlog.get_logger(__name__)

MODEL_VERSION = "all-MiniLM-L6-v2"

async def _get_connection() -> asyncpg.Connection:
    """Create a single connection to the database."""
    return await asyncpg.connect(settings.database_url)

async def store_embeddings(
    surfaces: list[str],
    embeddings: np.ndarray,
) -> int:
    """Store (surface, embedding) pairs in word_embeddings table.

    Uses INSERT ... ON CONFLICT (surface) DO UPDATE for idempotent upserts.

    Args:
        surfaces: List of N surface strings.
        embeddings: (N, 384) float32 array.

    Returns:
        Number of rows upserted.
    """
    if len(surfaces) != embeddings.shape[0]:
        raise SemanticError(f"surfaces ({len(surfaces)}) != embeddings rows ({embeddings.shape[0]})")

    conn = await _get_connection()
    try:
        count = 0
        for surface, emb in zip(surfaces, embeddings):
            vec_str = "[" + ",".join(f"{v:.8f}" for v in emb.tolist()) + "]"
            await conn.execute(
                """
                INSERT INTO word_embeddings (surface, embedding, model_version)
                VALUES ($1, $2::vector, $3)
                ON CONFLICT (surface) DO UPDATE
                SET embedding = EXCLUDED.embedding,
                    model_version = EXCLUDED.model_version,
                    created_at = NOW()
                """,
                surface, vec_str, MODEL_VERSION,
            )
            count += 1
        logger.info("semantic.persistence.stored", count=count)
        return count
    finally:
        await conn.close()

async def load_cached_embeddings(
    surfaces: list[str],
) -> dict[str, np.ndarray]:
    """Load cached embeddings for known surfaces.

    Args:
        surfaces: List of surface strings to look up.

    Returns:
        Dict mapping surface → (384,) float32 array. Missing surfaces omitted.
    """
    if not surfaces:
        return {}

    conn = await _get_connection()
    try:
        rows = await conn.fetch(
            "SELECT surface, embedding FROM word_embeddings WHERE surface = ANY($1::text[])",
            surfaces,
        )
        result = {}
        for row in rows:
            vec = np.array(row["embedding"], dtype=np.float32)
            result[row["surface"]] = vec
        logger.debug("semantic.persistence.loaded", requested=len(surfaces), found=len(result))
        return result
    finally:
        await conn.close()
```
